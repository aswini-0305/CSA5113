POLYALPHABETIC CIPHER - C PROGRAM

#include <stdio.h>
#include <string.h>
#include <ctype.h>

int main(void)
{
    char text[500], key[200];
    int i, j = 0, keylen;

    printf("Enter plaintext (no spaces): ");
    scanf("%499s", text);

    printf("Enter keyword: ");
    scanf("%199s", key);

    keylen = (int)strlen(key);

    if (keylen == 0)
        return 1;

    for (i = 0; text[i]; i++)
    {
        if (isalpha((unsigned char)text[i]))
        {
            int base = isupper((unsigned char)text[i]) ? 'A' : 'a';
            int shift = toupper((unsigned char)key[j % keylen]) - 'A';

            text[i] = (char)(base + (text[i] - base + shift) % 26);
            j++;
        }
    }

    printf("Encrypted text: %s\n", text);
    return 0;
}
