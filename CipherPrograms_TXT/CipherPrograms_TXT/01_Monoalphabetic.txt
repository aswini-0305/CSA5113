MONOALPHABETIC CIPHER - C PROGRAM

#include <stdio.h>
#include <string.h>
#include <ctype.h>

int main(void)
{
    char text[500], key[27];
    int i;

    printf("Enter 26-letter key: ");
    scanf("%26s", key);

    if (strlen(key) != 26)
    {
        printf("Key must contain exactly 26 letters.\n");
        return 1;
    }

    printf("Enter plaintext (no spaces): ");
    scanf("%499s", text);

    for (i = 0; text[i] != '\0'; i++)
    {
        if (isupper((unsigned char)text[i]))
            text[i] = toupper((unsigned char)key[text[i] - 'A']);
        else if (islower((unsigned char)text[i]))
            text[i] = tolower((unsigned char)key[text[i] - 'a']);
    }

    printf("Encrypted text: %s\n", text);
    return 0;
}
