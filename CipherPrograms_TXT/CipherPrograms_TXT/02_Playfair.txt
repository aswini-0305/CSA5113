#include <stdio.h>
#include <string.h>
#include <ctype.h>

char matrix[5][5];

void make_matrix(const char *key)
{
    int used[26] = {0}, k = 0, i;
    char c;

    used['J' - 'A'] = 1;

    for (i = 0; key[i] != '\0'; i++)
    {
        c = toupper((unsigned char)key[i]);
        if (c == 'J') c = 'I';

        if (c >= 'A' && c <= 'Z' && !used[c - 'A'])
        {
            matrix[k / 5][k % 5] = c;
            used[c - 'A'] = 1;
            k++;
        }
    }

    for (c = 'A'; c <= 'Z'; c++)
    {
        if (!used[c - 'A'])
        {
            matrix[k / 5][k % 5] = c;
            k++;
        }
    }
}

void find_pos(char c, int *r, int *col)
{
    int i, j;
    if (c == 'J') c = 'I';

    for (i = 0; i < 5; i++)
        for (j = 0; j < 5; j++)
            if (matrix[i][j] == c)
            {
                *r = i;
                *col = j;
                return;
            }
}

int main(void)
{
    char key[200], input[500], clean[500], pairs[1000];
    int i, n = 0, p = 0, r1, c1, r2, c2;

    printf("Enter keyword: ");
    scanf("%199s", key);

    printf("Enter plaintext (no spaces): ");
    scanf("%499s", input);

    for (i = 0; input[i]; i++)
    {
        char c = toupper((unsigned char)input[i]);

        if (c >= 'A' && c <= 'Z')
        {
            if (c == 'J') c = 'I';
            clean[n++] = c;
        }
    }
    clean[n] = '\0';

    for (i = 0; i < n;)
    {
        char a = clean[i], b;

        if (i + 1 >= n)
        {
            b = 'X';
            i++;
        }
        else if (clean[i] == clean[i + 1])
        {
            b = 'X';
            i++;
        }
        else
        {
            b = clean[i + 1];
            i += 2;
        }

        pairs[p++] = a;
        pairs[p++] = b;
    }

    pairs[p] = '\0';
    make_matrix(key);

    for (i = 0; i < p; i += 2)
    {
        find_pos(pairs[i], &r1, &c1);
        find_pos(pairs[i + 1], &r2, &c2);

        if (r1 == r2)
        {
            pairs[i] = matrix[r1][(c1 + 1) % 5];
            pairs[i + 1] = matrix[r2][(c2 + 1) % 5];
        }
        else if (c1 == c2)
        {
            pairs[i] = matrix[(r1 + 1) % 5][c1];
            pairs[i + 1] = matrix[(r2 + 1) % 5][c2];
        }
        else
        {
            pairs[i] = matrix[r1][c2];
            pairs[i + 1] = matrix[r2][c1];
        }
    }

    printf("Encrypted text: %s\n", pairs);
    return 0;
}
